rm -f *.o events/*.o gui/*.o Software_Explorer/*.o system/*.o Runtime_sdk/*.o Runtime_sdk/sdk/*.o Runtime_sdk/components/*.o

nasm -f elf64 system/memcpy.asm -o system/memcpy_asm.o
nasm -f elf64 system/memset.asm -o system/memset_asm.o
nasm -f elf64 system/syscall.asm -o system/syscall_asm.o

# 1. Sistema e Runtime (A base do executável)

gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -I. -c system/string.c -o system/string.o

gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -I. -c system/fonts.c -o system/fonts.o

# ------------- Eventos inicio --------------------------
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -I. -c events/cursor_engine.c -o events/cursor_engine.o

# ------------- Eventos fim -----------------------------

# ------------- Complemento gui -------------------------

gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -I. -c gui/wm.c -o gui/wm.o
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -I. -c gui/gui.c -o gui/gui.o

# 2. Biblioteca Visual e RAD - COMPONENTES

gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -Isystem -I. -c Runtime_sdk/sdk/TMemo.c -o Runtime_sdk/sdk/TMemo.o
