# 1. Montagem do arquivo Assembly local de cópia rápida
nasm -f elf64 string.asm -o string_asm.o

# 2. Compilação do módulo principal em C (Com otimização máxima -O3 + SSE3)
gcc -O3 -msse3 -m64 -ffreestanding -fno-stack-protector -fno-pie -c explorer.c -o explorer.o

# 3. Linkagem estática final incluindo o novo objeto de 64 bits (string_asm.o)
ld -m elf_x86_64 -T user.ld -static -no-pie \
    ../system/crt0.o \
    ../system/string.o \
    ../system/malloc.o \
    ../system/fonts.o \
    ../system/graphics.o \
    ../events/input_events.o \
    ../events/cursor_engine.o \
    ../gui/gui.o \
    ../gui/wm.o \
    ../gui/z_order.o \
    ../gui/gui_interaction.o \
    ../system/memcpy_asm.o \
    ../system/memset_asm.o \
    ../system/syscall_asm.o \
    explorer.o \
    -o explorer.elf
    
     mcopy -o -i disco_sata.img explorer.elf ::/
